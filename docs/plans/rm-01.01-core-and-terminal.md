# Plano de RM-01.01 — Núcleo e terminal

> **Para agentes:** use a skill `superpowers:subagent-driven-development` (recomendado) ou `superpowers:executing-plans` para executar este plano tarefa por tarefa. Os passos usam caixas de seleção (`- [ ]`) para acompanhar o andamento.

**Objetivo:** entregar o comando `janus` com o núcleo, `list`, `new`, o modo interativo com `fzf`, a prévia, o campo de nome, `--hold` e os testes automatizados ([[roadmap|RM-01.01]]).

**Arquitetura:** o núcleo (`internal/core`) descobre a pasta, lista, lê o manifesto, resolve o nome e copia, sem tocar no terminal, e devolve erros tipados. `internal/messages` transforma erros em texto, e `internal/cli` transforma erros em códigos de saída. `internal/interactive` é o laço de escolha e nome, com o `fzf` atrás das interfaces `Selector` e `NamePrompt`. Os erros que não são do núcleo moram em `internal/failure`, para que `messages` e `cli` os conheçam sem se importarem.

**Stack:** Go 1.27, biblioteca padrão, `github.com/BurntSushi/toml` 1.6.0 ([[decisions#DE-18 — O manifesto é lido com BurntSushi/toml|DE-18]]); `fzf` em tempo de execução, só no modo interativo, e `bat` opcional na prévia.

**Fontes:** [[bootstrap]] (contrato), [[decisions]] (DE-16 a DE-22), [[core|requisitos do núcleo]], [[cli|requisitos da linha de comando]] e [[tui|requisitos do modo interativo]].

// em docs/specs/bootstrap.md ("Cenários de aceite"), tira fotos de pastas e põe programas
// falsos no PATH. Só os testes o importam.
package testfixture

import (
	"crypto/sha256"
	"errors"
	"fmt"
	"io/fs"
	"os"
	"path/filepath"
	"strconv"
	"strings"
	"testing"
)

// Manifest é o manifesto da pasta de teste: as três regras do exemplo da spec.
const Manifest = `[names]
"info.ai/ai-memory/*"     = ".ai-memory.toml"
"info.ai/agents/*"        = "AGENTS.md"

const Binary = "PK\x03\x04\x00\x00\x00\x00fatiar"

// IDs são os identificadores da pasta de teste, na ordem da listagem (FR-CORE-09).

}{

	{"Empty Bash", "#!/usr/bin/env bash\n", 0o755},
	{"Empty File", "", 0o644},

// New monta a pasta de teste numa pasta temporária, com o manifesto, e devolve o caminho.
func New(t testing.TB) string {
	t.Helper()
	dir := t.TempDir()
	for _, f := range files {
		Write(t, dir, f.rel, f.content, f.perm)
	}
	Write(t, dir, ".janus.toml", Manifest, 0o644)
	return dir
}

// Use monta a pasta de teste e aponta XDG_TEMPLATES_DIR para ela.
func Use(t testing.TB) string {
	t.Helper()
	dir := New(t)
	t.Setenv("XDG_TEMPLATES_DIR", dir)
	return dir
}

// Write cria rel dentro de dir, com as pastas que faltarem, aplica perm exatamente e
// devolve o caminho do arquivo.
func Write(t testing.TB, dir, rel, content string, perm fs.FileMode) string {
	t.Helper()
	path := filepath.Join(dir, rel)
	if err := os.MkdirAll(filepath.Dir(path), 0o755); err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(path, []byte(content), perm); err != nil {
		t.Fatal(err)
	}
	// WriteFile passa pela umask; o Chmod garante a permissão pedida.
	if err := os.Chmod(path, perm); err != nil {
		t.Fatal(err)
	}
	return path
}

// Read devolve o conteúdo de path.
func Read(t testing.TB, path string) string {
	t.Helper()
	data, err := os.ReadFile(path)
	if err != nil {
		t.Fatal(err)
	}
	return string(data)
}

// Snapshot descreve tudo o que há em dir, inclusive os itens ocultos: caminho, tipo e
// permissões, data de modificação e resumo do conteúdo. Duas fotos iguais mostram que
// nada foi escrito na pasta (NFR-CORE-01).
func Snapshot(t testing.TB, dir string) string {
	t.Helper()
	var b strings.Builder
	err := filepath.WalkDir(dir, func(path string, d fs.DirEntry, err error) error {
		if err != nil {
			return err
		}
		info, err := d.Info()
		if err != nil {
			return err
		}
		rel, err := filepath.Rel(dir, path)
		if err != nil {
			return err
		}
		fmt.Fprintf(&b, "%s %v %d", rel, info.Mode(), info.ModTime().UnixNano())
		if info.Mode().IsRegular() {
			data, err := os.ReadFile(path)
			if err != nil {
				return err
			}
			fmt.Fprintf(&b, " %x", sha256.Sum256(data))
		}
		b.WriteByte('\n')
		return nil
	})
	if err != nil {
		t.Fatal(err)
	}
	return b.String()
}

// Reply é a resposta de um programa falso a uma chamada.
type Reply struct {
	Stdout string
	Stderr string
	Code   int
}

// Call é uma chamada recebida por um programa falso.
type Call struct {
	Args  []string
	Stdin string
}

// Fake é um programa falso posto no começo do PATH.
type Fake struct {
	dir string
}

// fakeScript registra os argumentos, separados por NUL, e a entrada de cada chamada, e
// responde com a resposta de mesmo número. A pasta vem de $0, que é o caminho do script.
const fakeScript = `#!/bin/sh
dir=${0%/*}
n=0
[ -f "$dir/count" ] && read -r n < "$dir/count"
n=$((n + 1))
echo "$n" > "$dir/count"
for arg in "$@"; do printf '%s\0' "$arg"; done > "$dir/call-$n.args"
cat > "$dir/call-$n.stdin"
[ -f "$dir/reply-$n.out" ] && cat "$dir/reply-$n.out"
[ -f "$dir/reply-$n.err" ] && cat "$dir/reply-$n.err" >&2
code=0
[ -f "$dir/reply-$n.code" ] && read -r code < "$dir/reply-$n.code"
exit "$code"
`

// FakeProgram põe no começo do PATH um programa falso chamado name. A chamada N recebe
// replies[N-1]; as chamadas além delas saem com 0 e sem saída.
func FakeProgram(t testing.TB, name string, replies ...Reply) *Fake {
	t.Helper()
	dir := t.TempDir()
	for i, r := range replies {
		n := i + 1
		Write(t, dir, fmt.Sprintf("reply-%d.out", n), r.Stdout, 0o644)
		Write(t, dir, fmt.Sprintf("reply-%d.err", n), r.Stderr, 0o644)
		Write(t, dir, fmt.Sprintf("reply-%d.code", n), strconv.Itoa(r.Code), 0o644)
	}
	Write(t, dir, name, fakeScript, 0o755)
	t.Setenv("PATH", dir+string(os.PathListSeparator)+os.Getenv("PATH"))
	return &Fake{dir: dir}
}

// Calls devolve as chamadas recebidas, na ordem.
func (f *Fake) Calls(t testing.TB) []Call {
	t.Helper()
	var calls []Call
	for n := 1; ; n++ {
		args, err := os.ReadFile(filepath.Join(f.dir, fmt.Sprintf("call-%d.args", n)))
		if errors.Is(err, fs.ErrNotExist) {
			return calls
		}
		if err != nil {
			t.Fatal(err)
		}
		list := strings.Split(string(args), "\x00")
		calls = append(calls, Call{
			Args:  list[:len(list)-1], // o último NUL deixa um elemento vazio no fim
			Stdin: Read(t, filepath.Join(f.dir, fmt.Sprintf("call-%d.stdin", n))),
		})
	}
}

// Package core é o núcleo do Janus: descobre a pasta de modelos, lista os modelos, lê o
// manifesto, resolve o nome final e cria o arquivo. Não imprime, não lê do teclado e não
// conhece códigos de saída: devolve valores e erros tipados (NFR-CORE-02).
package core

import "strings"

// matchPattern diz se pattern casa o identificador id inteiro (FR-CORE-12). O único
// curinga é "*", que casa qualquer sequência sem "/", inclusive vazia; todo o resto é
// literal. path.Match foi rejeitado porque interpreta "?", "[" e "\".
//
// Como "*" não atravessa "/", cada "/" do identificador casa com um "/" literal do padrão:
// basta comparar trecho a trecho.
func matchPattern(pattern, id string) bool {
	patternParts := strings.Split(pattern, "/")
	idParts := strings.Split(id, "/")
	if len(patternParts) != len(idParts) {
		return false
	}
	for i := range patternParts {
		if !matchPart(patternParts[i], idParts[i]) {
			return false
		}
	}
	return true
}

// matchPart casa um trecho sem "/". Ao falhar, volta ao último "*" visto e faz ele engolir
// mais um byte: é o algoritmo clássico de curinga, sem recursão.
func matchPart(pattern, s string) bool {
	p, i := 0, 0
	star, mark := -1, 0
	for i < len(s) {
		switch {
		case p < len(pattern) && pattern[p] == '*':
			star, mark = p, i
			p++
		case p < len(pattern) && pattern[p] == s[i]:
			p++
			i++
		case star >= 0:
			mark++
			i = mark
			p = star + 1
		default:
			return false
		}
	}
	for p < len(pattern) && pattern[p] == '*' {
		p++
	}
	return p == len(pattern)
}

package core

import (
	"fmt"
	"strings"
	"unicode"
)

// NameProblem diz por que um nome é inválido.
type NameProblem int

const (
	NameEmpty   NameProblem = iota + 1 // vazio depois de aparar as pontas
	NameDots                           // "." ou ".."
	NameSlash                          // contém "/"
	NameControl                        // contém caractere de controle
)

// InvalidNameError é um nome inválido (código 7).
type InvalidNameError struct {
	Name    string // o nome como foi informado
	Problem NameProblem
}

func (e *InvalidNameError) Error() string {
	return fmt.Sprintf("invalid name %q (problem %d)", e.Name, e.Problem)
}

// NameRequiredError é um modelo ask sem nome informado, fora do modo interativo (código 6).
type NameRequiredError struct {
	ID string
}

func (e *NameRequiredError) Error() string {
	return fmt.Sprintf("template %q needs a name", e.ID)
}

// cleanName apara os espaços das pontas e valida o nome (FR-CORE-16). Só o espaço é
// aparado: uma tabulação na ponta fica e torna o nome inválido, como caractere de controle.
func cleanName(name string) (string, error) {
	n := strings.Trim(name, " ")
	var problem NameProblem
	switch {
	case n == "":
		problem = NameEmpty
	case n == "." || n == "..":
		problem = NameDots
	case strings.Contains(n, "/"):
		problem = NameSlash
	case strings.ContainsFunc(n, unicode.IsControl):
		problem = NameControl
	default:
		return n, nil
	}
	return "", &InvalidNameError{Name: name, Problem: problem}
}

// extension devolve a extensão de name, com o ponto, ou "" (FR-CORE-17): o trecho a
// partir do último ponto, se o ponto não é o primeiro caractere, há algo depois dele e o
// trecho não tem espaço. Extensão composta perde a primeira parte (DT-08).
func extension(name string) string {
	i := strings.LastIndexByte(name, '.')
	if i <= 0 || i == len(name)-1 || strings.Contains(name[i:], " ") {
		return ""
	}
	return name[i:]
}

// FinalName devolve o nome com que o arquivo de t vai nascer. input é o nome informado
// pelo usuário e só vale com informed. Sem ele, vale o nome fixo, e um modelo ask falha
// com NameRequiredError (FR-CLI-08). Com ele, o nome é validado e ganha a extensão de
// referência quando ela falta, mesmo num modelo fixed (FR-CORE-16, FR-CORE-17).
func FinalName(t Template, input string, informed bool) (string, error) {
	if !informed {
		if t.Mode == NameFixed {
			return t.FixedName, nil
		}
		return "", &NameRequiredError{ID: t.ID}
	}
	name, err := cleanName(input)
	if err != nil {
		return "", err
	}
	ext := t.ReferenceExtension()
	if ext == "" || hasSuffixFold(name, ext) {
		return name, nil
	}
	return name + ext, nil
}

// hasSuffixFold diz se s termina com suffix, sem diferenciar maiúsculas.
func hasSuffixFold(s, suffix string) bool {
	return len(s) >= len(suffix) && strings.EqualFold(s[len(s)-len(suffix):], suffix)
}

package core

import (
	"errors"
	"fmt"
	"io/fs"
	"os"
	"path/filepath"
	"strings"

	"github.com/BurntSushi/toml"
)

// ManifestFile é o nome do manifesto, na raiz da pasta de modelos.
const ManifestFile = ".janus.toml"

// Rule é uma regra do manifesto: um padrão e o nome fixo que ele dá.
type Rule struct {
	Pattern string
	Name    string
}

// Manifest guarda as regras na ordem em que aparecem no arquivo.
type Manifest struct {
	Path  string
	Rules []Rule
}

// ManifestProblem diz o que torna o manifesto inválido.
type ManifestProblem int

const (
	ManifestSyntax        ManifestProblem = iota + 1 // TOML malformado, inclusive chave repetida
	ManifestUnknownKey                               // chave ou tabela fora de [names]
	ManifestNamesNotTable                            // names existe, mas não é uma tabela
	ManifestNotString                                // valor de regra que não é texto
	ManifestDoubleStar                               // padrão com "**", reservado
	ManifestInvalidName                              // nome fixo inválido; Err traz o InvalidNameError
	ManifestAmbiguous                                // dois ou mais padrões com "*" casam o mesmo modelo
)

// ManifestError é o manifesto inválido (código 4).
type ManifestError struct {
	Path     string
	Problem  ManifestProblem
	Key      string   // a chave ou o padrão envolvido
	Line     int      // a linha, no erro de sintaxe
	ID       string   // o modelo, em ManifestAmbiguous
	Patterns []string // os padrões em conflito, na ordem do arquivo
	Err      error    // a causa, quando houver
}

func (e *ManifestError) Error() string {
	return fmt.Sprintf("invalid manifest %s (problem %d, key %q)", e.Path, e.Problem, e.Key)
}

func (e *ManifestError) Unwrap() error { return e.Err }

// LoadManifest lê o manifesto da pasta de modelos dir. Arquivo ausente, vazio ou sem a
// tabela [names] é um manifesto sem regras (FR-CORE-10).
//
// O TOML é lido como tabela genérica e conferido chave a chave, na ordem do arquivo
// (MetaData.Keys), para a mensagem citar sempre a primeira chave errada. Ler direto numa
// struct foi rejeitado: o erro de tipo da biblioteca não diz a chave de forma estruturada.
func LoadManifest(dir string) (Manifest, error) {
	path := filepath.Join(dir, ManifestFile)
	data, err := os.ReadFile(path)
	if errors.Is(err, fs.ErrNotExist) {
		return Manifest{Path: path}, nil
	}
	if err != nil {
		return Manifest{}, err
	}
	var raw map[string]any
	md, err := toml.Decode(string(data), &raw)
	if err != nil {
		e := &ManifestError{Path: path, Problem: ManifestSyntax, Err: err}
		var parseErr toml.ParseError
		if errors.As(err, &parseErr) {
			e.Line = parseErr.Position.Line
		}
		return Manifest{}, e
	}
	return parseRules(path, raw, md.Keys())
}

// parseRules confere a estrutura do manifesto e extrai as regras (FR-CORE-11, FR-CORE-12,
// FR-CORE-14).
func parseRules(path string, raw map[string]any, keys []toml.Key) (Manifest, error) {
	fail := func(problem ManifestProblem, key string, cause error) (Manifest, error) {
		return Manifest{}, &ManifestError{Path: path, Problem: problem, Key: key, Err: cause}
	}
	for _, k := range keys {
		if k[0] != "names" {
			return fail(ManifestUnknownKey, k[0], nil)
		}
	}
	m := Manifest{Path: path}
	value, ok := raw["names"]
	if !ok {
		return m, nil
	}
	names, ok := value.(map[string]any)
	if !ok {
		return fail(ManifestNamesNotTable, "names", nil)
	}
	seen := map[string]bool{}
	for _, k := range keys {
		if len(k) < 2 || seen[k[1]] {
			continue
		}
		pattern := k[1]
		seen[pattern] = true
		name, ok := names[pattern].(string)
		if !ok {
			return fail(ManifestNotString, pattern, nil)
		}
		if strings.Contains(pattern, "**") {
			return fail(ManifestDoubleStar, pattern, nil)
		}
		clean, err := cleanName(name)
		if err != nil {
			return fail(ManifestInvalidName, pattern, err)
		}
		m.Rules = append(m.Rules, Rule{Pattern: pattern, Name: clean})
	}
	return m, nil
}

```go
package core

import (
	"fmt"
	"os"
	"path/filepath"
	"strings"
)

// TemplatesDirSource diz de onde veio a pasta de modelos.
type TemplatesDirSource int

const (
	SourceEnv      TemplatesDirSource = iota + 1 // a variável XDG_TEMPLATES_DIR
	SourceUserDirs                               // o arquivo user-dirs.dirs
)

// TemplatesDirReason diz por que a pasta de modelos não foi encontrada.
type TemplatesDirReason int

const (
	DirFileMissing TemplatesDirReason = iota + 1 // user-dirs.dirs ausente ou ilegível
	DirLineMissing                               // user-dirs.dirs sem XDG_TEMPLATES_DIR
	DirRelative                                  // valor que não é absoluto nem começa por $HOME
	DirNotDir                                    // caminho inexistente ou que não é pasta
	DirIsHome                                    // a pasta pessoal: pasta de modelos desativada
	DirNoHome                                    // HOME vazia, sem XDG_TEMPLATES_DIR
)

// TemplatesDirError é a pasta de modelos não encontrada (código 3).
type TemplatesDirError struct {
	Source TemplatesDirSource
	Reason TemplatesDirReason
	File   string // o user-dirs.dirs, quando Source é SourceUserDirs
	Value  string // o valor ou o caminho tentado
	Err    error  // a causa, quando houver
}

func (e *TemplatesDirError) Error() string {
	return fmt.Sprintf("templates dir not found (source %d, reason %d): %q", e.Source, e.Reason, e.Value)
}

func (e *TemplatesDirError) Unwrap() error { return e.Err }

// FindTemplatesDir descobre a pasta de modelos (FR-CORE-01 a FR-CORE-04). getenv lê o
// ambiente; os testes passam um mapa, sem tocar no ambiente do processo.
func FindTemplatesDir(getenv func(string) string) (string, error) {
	home := getenv("HOME")
	if value := getenv("XDG_TEMPLATES_DIR"); value != "" {
		// Sem segunda tentativa: um valor ruim falha, mesmo com um user-dirs.dirs válido.
		return checkTemplatesDir(value, home, &TemplatesDirError{Source: SourceEnv})
	}
	fail := &TemplatesDirError{Source: SourceUserDirs}
	if home == "" {
		fail.Reason = DirNoHome
		return "", fail
	}
	config := getenv("XDG_CONFIG_HOME")
	if config == "" {
		config = filepath.Join(home, ".config")
	}
	fail.File = filepath.Join(config, "user-dirs.dirs")
	data, err := os.ReadFile(fail.File)
	if err != nil {
		fail.Reason, fail.Err = DirFileMissing, err
		return "", fail
	}
	value, ok := userDirsValue(string(data))
	if !ok {
		fail.Reason = DirLineMissing
		return "", fail
	}
	dir, ok := expandHome(value, home)
	if !ok {
		fail.Reason, fail.Value = DirRelative, value
		return "", fail
	}
	return checkTemplatesDir(dir, home, fail)
}

// userDirsValue devolve o valor de XDG_TEMPLATES_DIR em user-dirs.dirs, lido como a GLib
// lê, porque é ela que dá a pasta ao Nemo (DE-20): espaços no início e em volta do "=",
// a última linha vence, e o valor vai da primeira à última aspa, sem desfazer escapes.
func userDirsValue(content string) (string, bool) {
	value, found := "", false
	for line := range strings.Lines(content) {
		rest, ok := strings.CutPrefix(strings.TrimLeft(line, " \t"), "XDG_TEMPLATES_DIR")
		if !ok {
			continue
		}
		rest, ok = strings.CutPrefix(strings.TrimLeft(rest, " \t"), "=")
		if !ok {
			continue
		}
		rest, ok = strings.CutPrefix(strings.TrimLeft(rest, " \t"), `"`)
		end := strings.LastIndexByte(rest, '"')
		if !ok || end < 0 {
			continue
		}
		value, found = rest[:end], true
	}
	return value, found
}

// expandHome expande o $HOME do começo do valor. Como na GLib, só "$HOME" sozinho ou
// seguido de "/" é aceito; um valor que não começa por "$HOME" precisa ser absoluto.
func expandHome(value, home string) (string, bool) {
	rest, ok := strings.CutPrefix(value, "$HOME")
	if !ok {
		return value, strings.HasPrefix(value, "/")
	}
	if rest != "" && rest[0] != '/' {
		return "", false
	}
	return filepath.Join(home, rest), true
}

// checkTemplatesDir confere que dir é absoluto, não é a pasta pessoal e é uma pasta. As
// barras do fim saem antes da comparação.
func checkTemplatesDir(dir, home string, fail *TemplatesDirError) (string, error) {
	fail.Value = dir
	if !filepath.IsAbs(dir) {
		fail.Reason = DirRelative
		return "", fail
	}
	dir = filepath.Clean(dir)
	fail.Value = dir
	if home != "" && dir == filepath.Clean(home) {
		fail.Reason = DirIsHome
		return "", fail
	}
	// Stat segue o link: ~/Modelos é um link na máquina do autor (FR-CORE-04).
	info, err := os.Stat(dir)
	if err != nil || !info.IsDir() {
		fail.Reason, fail.Err = DirNotDir, err
		return "", fail
	}
	return dir, nil
}

import (
	"fmt"
	"io/fs"
	"os"
	"path"
	"path/filepath"
	"slices"
	"strings"
	"unicode"
)

// Catalog é a pasta de modelos lida: os modelos, em ordem, com o modo de nome resolvido.
type Catalog struct {
	Dir       string
	Templates []Template
}

// TemplateNotFoundError é um identificador que a listagem não traz (código 5).
type TemplateNotFoundError struct {
	ID string
}

func (e *TemplateNotFoundError) Error() string {
	return fmt.Sprintf("template %q not found", e.ID)
}

// LoadCatalog lê o manifesto e a pasta de modelos dir e resolve a regra de cada modelo.
// O manifesto é lido antes da pasta para que um manifesto malformado (código 4) vença uma
// falha de leitura da pasta (código 1), como pede a ordem de verificação (FR-CLI-11).
func LoadCatalog(dir string) (Catalog, error) {
	m, err := LoadManifest(dir)
	if err != nil {
		return Catalog{}, err
	}
	templates, err := listTemplates(dir)
	if err != nil {
		return Catalog{}, err
	}
	for i := range templates {
		if err := m.resolve(&templates[i]); err != nil {
			return Catalog{}, err
		}
	}
	return Catalog{Dir: dir, Templates: templates}, nil
}

// IDs devolve os identificadores, na ordem da listagem.
func (c Catalog) IDs() []string {
	ids := make([]string, len(c.Templates))
	for i, t := range c.Templates {
		ids[i] = t.ID
	}
	return ids
}

// Find procura o modelo pelo identificador, só na listagem. O identificador nunca é
// juntado à pasta para formar um caminho, então "../x" ou ".janus.toml" não são modelos
// (FR-CORE-15).
func (c Catalog) Find(id string) (Template, error) {
	for _, t := range c.Templates {
		if t.ID == id {
			return t, nil
		}
	}
	return Template{}, &TemplateNotFoundError{ID: id}
}

// listTemplates percorre a pasta de modelos e devolve os modelos em ordem, todos em modo
// ask (FR-CORE-05 a FR-CORE-09).
func listTemplates(root string) ([]Template, error) {
	var templates []Template
	var walk func(category string) error
	walk = func(category string) error {
		// os.ReadDir segue a própria pasta quando ela é um link (FR-CORE-04); as entradas vêm
		// com o tipo do link, e não do alvo.
		entries, err := os.ReadDir(filepath.Join(root, category))
		if err != nil {
			return err
		}
		for _, e := range entries {
			name := e.Name()
			if strings.HasPrefix(name, ".") || strings.ContainsFunc(name, unicode.IsControl) {
				continue // FR-CORE-06, FR-CORE-08
			}
			id := path.Join(category, name)
			switch {
			case e.IsDir():
				if err := walk(id); err != nil {
					return err
				}
			case e.Type()&fs.ModeSymlink != 0:
				// Link para arquivo é modelo; quebrado é ignorado; link para pasta não é
				// seguido, o que também evita laços (FR-CORE-07).
				if info, err := os.Stat(filepath.Join(root, id)); err == nil && info.Mode().IsRegular() {
					templates = append(templates, newTemplate(root, category, name))
				}
			case e.Type().IsRegular():
				templates = append(templates, newTemplate(root, category, name))
			}
		}
		return nil
	}
	if err := walk(""); err != nil {
		return nil, err
	}
	slices.SortFunc(templates, func(a, b Template) int {
		if c := strings.Compare(strings.ToLower(a.ID), strings.ToLower(b.ID)); c != 0 {
			return c
		}
		return strings.Compare(a.ID, b.ID)
	})
	return templates, nil
}

func newTemplate(root, category, file string) Template {
	id := path.Join(category, file)
	return Template{ID: id, Category: category, File: file, Path: filepath.Join(root, id), Mode: NameAsk}
}

// resolve aplica as regras do manifesto a t (FR-CORE-13). Um padrão sem "*" igual ao
// identificador vence; senão, conta quantos padrões com "*" casam.
func (m Manifest) resolve(t *Template) error {
	var matches []Rule
	for _, r := range m.Rules {
		if !strings.Contains(r.Pattern, "*") {
			if r.Pattern == t.ID {
				t.Mode, t.FixedName = NameFixed, r.Name
				return nil
			}
			continue
		}
		if matchPattern(r.Pattern, t.ID) {
			matches = append(matches, r)
		}
	}
	switch len(matches) {
	case 0:
		return nil
	case 1:
		t.Mode, t.FixedName = NameFixed, matches[0].Name
		return nil
	}
	patterns := make([]string, len(matches))
	for i, r := range matches {
		patterns[i] = r.Pattern
	}
	return &ManifestError{Path: m.Path, Problem: ManifestAmbiguous, ID: t.ID, Patterns: patterns}
}

package core

import (
	"errors"
	"fmt"
	"io"
	"io/fs"
	"os"
	"path/filepath"
)

// DestinationDirError é a pasta de destino inexistente ou que não é pasta (código 10).
type DestinationDirError struct {
	Dir string // a pasta como foi informada
	Err error
}

func (e *DestinationDirError) Error() string {
	return fmt.Sprintf("destination dir %q not found", e.Dir)
}

func (e *DestinationDirError) Unwrap() error { return e.Err }

// DestinationExistsError é o destino que já existe (código 8).
type DestinationExistsError struct {
	Path string
}

func (e *DestinationExistsError) Error() string {
	return fmt.Sprintf("destination %s already exists", e.Path)
}

// DestinationDir confere a pasta de destino e devolve o caminho absoluto dela. Um link
// para pasta é aceito (FR-CORE-18). O caminho vazio é recusado antes do filepath.Abs, que
// o trataria como a pasta atual.
func DestinationDir(dir string) (string, error) {
	if dir == "" {
		return "", &DestinationDirError{Dir: dir}
	}
	abs, err := filepath.Abs(dir)
	if err != nil {
		return "", &DestinationDirError{Dir: dir, Err: err}
	}
	info, err := os.Stat(abs)
	if err != nil {
		return "", &DestinationDirError{Dir: dir, Err: err}
	}
	if !info.IsDir() {
		return "", &DestinationDirError{Dir: dir}
	}
	return abs, nil
}

// copyContent copia o conteúdo do modelo. É variável só para o teste simular uma falha
// no meio da cópia (FR-CORE-22).
var copyContent = io.Copy

// Create cria name em destDir com o conteúdo e as permissões de t e devolve o caminho
// criado. destDir deve vir de DestinationDir.
//
// A criação é exclusiva (O_EXCL): falha se existir qualquer coisa com o nome, até um link
// quebrado, e fecha a janela entre checar e criar (FR-CORE-19). Checar antes com Stat foi
// rejeitado por deixar essa janela aberta. O destino é aberto antes do modelo para que
// "destino existe" (8) vença uma falha de leitura (1), como pede a ordem de verificação.
func Create(t Template, destDir, name string) (string, error) {
	dest := filepath.Join(destDir, name)
	out, err := os.OpenFile(dest, os.O_WRONLY|os.O_CREATE|os.O_EXCL, 0o600)
	if errors.Is(err, fs.ErrExist) {
		return "", &DestinationExistsError{Path: dest}
	}
	if err != nil {
		return "", err
	}
	if err := fill(out, t.Path); err != nil {
		out.Close()
		os.Remove(dest) // FR-CORE-22
		return "", err
	}
	if err := out.Close(); err != nil {
		os.Remove(dest)
		return "", err
	}
	return dest, nil
}

// fill copia o modelo src para out, byte a byte (FR-CORE-20), e aplica as permissões dele.
// O Chmod no arquivo aberto aplica as permissões exatas; as do OpenFile passam pela umask
// (FR-CORE-21). Como o modelo é aberto com os.Open, um link é seguido até o alvo.
func fill(out *os.File, src string) error {
	in, err := os.Open(src)
	if err != nil {
		return err
	}
	defer in.Close()
	info, err := in.Stat()
	if err != nil {
		return err
	}
	if _, err := copyContent(out, in); err != nil {
		return err
	}
	return out.Chmod(info.Mode().Perm())
}

import (
	"bytes"
	"errors"
	"io"
	"os"

// PreviewKind diz como o modelo aparece na prévia.
type PreviewKind int

const (
	PreviewText   PreviewKind = iota + 1 // conteúdo mostrado
	PreviewEmpty                         // aviso de arquivo vazio
	PreviewBinary                        // aviso de arquivo binário, com o tamanho
)

// binarySniffLen é quantos bytes do começo são examinados à procura de um byte nulo, o
// mesmo critério do git.
const binarySniffLen = 8000

// Inspect classifica o arquivo path para a prévia e devolve o tamanho dele (FR-CORE-23).
func Inspect(path string) (PreviewKind, int64, error) {
	f, err := os.Open(path)
	if err != nil {
		return 0, 0, err
	}
	defer f.Close()
	info, err := f.Stat()
	if err != nil {
		return 0, 0, err
	}
	if info.Size() == 0 {
		return PreviewEmpty, 0, nil
	}
	head := make([]byte, binarySniffLen)
	n, err := io.ReadFull(f, head)
	if err != nil && !errors.Is(err, io.ErrUnexpectedEOF) {
		return 0, 0, err
	}
	if bytes.IndexByte(head[:n], 0) >= 0 {
		return PreviewBinary, info.Size(), nil
	}
	return PreviewText, info.Size(), nil
}

// quoteAll cita os itens entre aspas e os junta à moda brasileira: "a", "b" e "c".
func quoteAll(items []string) string {
	quoted := make([]string, len(items))
	for i, s := range items {
		quoted[i] = strconv.Quote(s)
	}
	if len(quoted) < 2 {
		return strings.Join(quoted, "")
	}
	return strings.Join(quoted[:len(quoted)-1], ", ") + " e " + quoted[len(quoted)-1]
}

// formatSize escreve um tamanho do jeito brasileiro: "24,0 kB".
func formatSize(n int64) string {
	switch {
	case n == 1:
		return "1 byte"
	case n < 1024:
		return fmt.Sprintf("%d bytes", n)
	case n < 1024*1024:
		return decimal(float64(n)/1024) + " kB"
	}
	return decimal(float64(n)/(1024*1024)) + " MB"
}

func decimal(f float64) string {
	return strings.Replace(strconv.FormatFloat(f, 'f', 1, 64), ".", ",", 1)
}

// Package cli é a linha de comando: lê os argumentos, roda list, new e o modo interativo e
// traduz os erros em códigos de saída. É o contrato de que os adaptadores dependem.
package cli

import (
	"strings"

	"gitlab.com/rg-dev/janus/internal/failure"
)

// parsed é o resultado da leitura dos argumentos de um nível da linha de comando.
type parsed struct {
	options map[string]string // opção, sem os hífens, e o valor; "" nas que não recebem valor
	args    []string          // os argumentos posicionais, na ordem
}

func (p parsed) has(option string) bool {
	_, ok := p.options[option]
	return ok
}

// parseArgs lê args conforme options, que diz quais opções existem e se recebem valor
// (DE-19). Aceita "--opcao valor" e "--opcao=valor" em qualquer posição; "--" encerra as
// opções. Opção desconhecida, com um hífen só, repetida ou sem valor é uso incorreto
// (FR-CLI-01, FR-CLI-02). O pacote flag foi rejeitado porque para no primeiro argumento
// posicional, e o contrato põe as opções depois do identificador.
func parseArgs(command string, args []string, options map[string]bool) (parsed, error) {
	usage := func(problem failure.UsageProblem, arg string) (parsed, error) {
		return parsed{}, &failure.UsageError{Problem: problem, Arg: arg, Command: command}
	}
	p := parsed{options: map[string]string{}}
	for i := 0; i < len(args); i++ {
		arg := args[i]
		switch {
		case arg == "--":
			p.args = append(p.args, args[i+1:]...)
			return p, nil
		case strings.HasPrefix(arg, "--"):
			name, value, hasValue := strings.Cut(arg[2:], "=")
			takesValue, known := options[name]
			if !known || (hasValue && !takesValue) {
				return usage(failure.UnknownOption, arg)
			}
			if p.has(name) {
				return usage(failure.RepeatedOption, "--"+name)
			}
			if takesValue && !hasValue {
				if i+1 == len(args) {
					return usage(failure.MissingValue, arg)
				}
				i++
				value = args[i]
			}
			p.options[name] = value
		case strings.HasPrefix(arg, "-"):
			return usage(failure.UnknownOption, arg)
		default:
			p.args = append(p.args, arg)
		}
	}
	return p, nil
}

// Package interactive é o modo interativo: o seletor, a prévia e o campo de nome. Ele chama
// o núcleo no mesmo processo, nunca o próprio binário. O seletor e o campo ficam atrás de
// interfaces para os testes os substituírem (NFR-TUI-03).
package interactive

import (
	"errors"

	"gitlab.com/rg-dev/janus/internal/core"
	"gitlab.com/rg-dev/janus/internal/failure"
	"gitlab.com/rg-dev/janus/internal/messages"
)

// Selector escolhe um modelo entre os identificadores. Cancelar devolve failure.ErrCanceled.
type Selector interface {
	Select(ids []string) (string, error)
}

// NamePrompt pede o nome do arquivo a partir de um valor inicial editável, mostrando o
// aviso quando houver. Cancelar devolve failure.ErrCanceled.
type NamePrompt interface {
	Ask(initial, warning string) (string, error)
}

// Deps é o que o modo interativo usa de fora.
type Deps struct {
	Getenv     func(string) string
	Getwd      func() (string, error)
	LookPath   func(string) (string, error)
	IsTerminal bool // a entrada padrão e a saída de erro são um terminal
	Selector   Selector
	Prompt     NamePrompt
}

// Run escolhe um modelo, cria o arquivo na pasta atual e devolve o caminho criado. As
// verificações seguem a ordem do contrato: 2, 3, 4, 5 e 9 (FR-TUI-02).
func Run(d Deps) (string, error) {
	if !d.IsTerminal {
		return "", &failure.UsageError{Problem: failure.NoTerminal}
	}
	dir, err := core.FindTemplatesDir(d.Getenv)
	if err != nil {
		return "", err
	}
	cat, err := core.LoadCatalog(dir)
	if err != nil {
		return "", err
	}
	if len(cat.Templates) == 0 {
		return "", &failure.NoTemplatesError{Dir: dir}
	}
	if _, err := d.LookPath("fzf"); err != nil {
		return "", &failure.MissingProgramError{Program: "fzf"}
	}
	cwd, err := d.Getwd()
	if err != nil {
		return "", err
	}
	destDir, err := core.DestinationDir(cwd)
	if err != nil {
		return "", err
	}
	id, err := d.Selector.Select(cat.IDs())
	if err != nil {
		return "", err
	}
	t, err := cat.Find(id)
	if err != nil {
		return "", err
	}
	return create(t, destDir, d.Prompt)
}

// create tenta criar o arquivo e, quando o nome digitado é inválido ou o destino já existe,
// avisa e reabre o campo com o nome que falhou, até dar certo ou o usuário cancelar
// (FR-TUI-04 a FR-TUI-07). Um modelo fixed não pergunta na primeira tentativa.
func create(t core.Template, destDir string, prompt NamePrompt) (string, error) {
	input, warning := t.File, ""
	asking := t.Mode == core.NameAsk
	for {
		if asking {
			typed, err := prompt.Ask(input, warning)
			if err != nil {
				return "", err
			}
			input = typed
		}
		name, err := core.FinalName(t, input, asking)
		// Só o nome digitado pode ser inválido: o nome fixo já foi validado com o manifesto.
		if asking && errors.As(err, new(*core.InvalidNameError)) {
			warning = messages.Describe(err)
			continue
		}
		if err != nil {
			return "", err
		}
		path, err := core.Create(t, destDir, name)
		if errors.As(err, new(*core.DestinationExistsError)) {
			input, warning, asking = name, messages.Describe(err), true
			continue
		}
		return path, err
	}
}

package interactive

import (
	"errors"
	"io"
	"os/exec"
	"strings"

	"gitlab.com/rg-dev/janus/internal/failure"
	"gitlab.com/rg-dev/janus/internal/messages"
)

// FzfSelector é o seletor do terminal: o fzf, com a prévia do próprio janus (DE-06, DE-12).
type FzfSelector struct {
	Executable string    // o próprio janus, que a prévia chama
	Stderr     io.Writer // o terminal; recebe as mensagens de erro do fzf
}

// Select mostra ids no fzf e devolve o escolhido. Cancelar, ou confirmar sem nenhum
// resultado, devolve failure.ErrCanceled (FR-TUI-10).
func (s FzfSelector) Select(ids []string) (string, error) {
	out, code, err := runFzf(s.Stderr, strings.Join(ids, "\n")+"\n", selectorArgs(s.Executable))
	if err != nil {
		return "", err
	}
	id := firstLine(out)
	if code != 0 || id == "" {
		return "", failure.ErrCanceled
	}
	return id, nil
}

// selectorArgs monta as opções do seletor. Elas vêm depois das do FZF_DEFAULT_OPTS e
// prevalecem, então o tema do usuário fica e o contrato não muda.
func selectorArgs(executable string) []string {
	return []string{
		// Anulam opções do usuário que mudariam a saída (NFR-TUI-02).
		"--no-multi", "--no-print-query", "--no-expect",
		// A prévia roda com sh, e não com o $SHELL do usuário: o fish tem outras regras de
		// aspas, e o {} do fzf sai citado para sh (NFR-TUI-01). O identificador nunca é
		// juntado ao caminho aqui: quem o transforma em arquivo é o subcomando (DE-12).
		"--with-shell", "sh -c",
		"--preview", shellQuote(executable) + " " + PreviewCommand + " {}",
		"--prompt=" + messages.SelectorLabel,
	}
}

// FzfPrompt é o campo de nome: o fzf com a lista vazia (DE-17).
type FzfPrompt struct {
	Stderr io.Writer // o terminal; recebe as mensagens de erro do fzf
}

// Ask abre o campo com initial e, quando há, o aviso no cabeçalho. Devolve o texto
// confirmado; Esc, Ctrl-C e Ctrl-D devolvem failure.ErrCanceled (FR-TUI-11).
func (p FzfPrompt) Ask(initial, warning string) (string, error) {
	out, code, err := runFzf(p.Stderr, "", promptArgs(initial, warning))
	if err != nil {
		return "", err
	}
	if code == 130 {
		return "", failure.ErrCanceled
	}
	// Com a lista vazia, Enter sai com 1 (nenhum resultado); --print-query põe o texto na
	// primeira linha mesmo assim.
	return firstLine(out), nil
}

func promptArgs(initial, warning string) []string {
	args := []string{
		"--no-multi", "--no-expect", "--print-query",
		"--query=" + initial,
		"--prompt=" + messages.NameLabel,
		// Na configuração padrão, Ctrl-D só cancela com o campo vazio; a spec pede que
		// cancele sempre (DE-17).
		"--bind=ctrl-d:abort",
		"--info=hidden",
		"--height=~6",
	}
	if warning != "" {
		args = append(args, "--header="+warning)
	}
	return args
}

// runFzf roda o fzf com input na entrada e devolve a saída padrão e o código de saída. O
// fzf desenha no /dev/tty, então a saída de erro dele vai direto para o terminal, com as
// mensagens de erro que ele mesmo escreve. Os códigos 0, 1 e 130 são respostas; qualquer
// outro é falha (FR-TUI-10).
func runFzf(stderr io.Writer, input string, args []string) (string, int, error) {
	cmd := exec.Command("fzf", args...)
	cmd.Stdin = strings.NewReader(input)
	var out strings.Builder
	cmd.Stdout = &out
	cmd.Stderr = stderr
	err := cmd.Run()
	var exit *exec.ExitError
	if errors.As(err, &exit) {
		code := exit.ExitCode()
		if code == 1 || code == 130 {
			return out.String(), code, nil
		}
		return "", 0, &failure.ProgramError{Program: "fzf", Code: code}
	}
	return out.String(), 0, err
}

func firstLine(s string) string {
	line, _, _ := strings.Cut(s, "\n")
	return line
}

// shellQuote cita s entre aspas simples, para sh.
func shellQuote(s string) string {
	return "'" + strings.ReplaceAll(s, "'", `'\''`) + "'"
}

import (
	"fmt"
	"io"
	"os"
	"os/exec"

	"gitlab.com/rg-dev/janus/internal/core"
	"gitlab.com/rg-dev/janus/internal/messages"

// PreviewCommand é o subcomando interno que o fzf chama para a prévia, com o
// identificador. Não aparece na ajuda e não é contrato (DE-12).
const PreviewCommand = "__preview"

// PreviewDeps é o que a prévia usa de fora.
type PreviewDeps struct {
	Getenv   func(string) string
	LookPath func(string) (string, error)
	Stdout   io.Writer
}

// Preview escreve a prévia do modelo id (FR-TUI-12): texto realçado pelo bat quando ele
// está no PATH, ou cru; uma linha de aviso para binário e para vazio. O identificador
// passa pela listagem, como em new; nunca vira caminho direto (FR-CORE-15).
func Preview(id string, d PreviewDeps) error {
	dir, err := core.FindTemplatesDir(d.Getenv)
	if err != nil {
		return err
	}
	cat, err := core.LoadCatalog(dir)
	if err != nil {
		return err
	}
	t, err := cat.Find(id)
	if err != nil {
		return err
	}
	kind, size, err := core.Inspect(t.Path)
	if err != nil {
		return err
	}
	switch kind {
	case core.PreviewEmpty:
		_, err = fmt.Fprintln(d.Stdout, messages.PreviewEmpty)
		return err
	case core.PreviewBinary:
		_, err = fmt.Fprintln(d.Stdout, messages.PreviewBinary(size))
		return err
	}
	if bat, err := d.LookPath("bat"); err == nil {
		// O caminho vai como argumento, sem shell: nomes com espaço e ";" chegam intactos.
		cmd := exec.Command(bat, "--color=always", "--style=plain", "--paging=never", t.Path)
		cmd.Stdout = d.Stdout
		cmd.Stderr = d.Stdout
		return cmd.Run()
	}
	f, err := os.Open(t.Path)
	if err != nil {
		return err
	}
	defer f.Close()
	_, err = io.Copy(d.Stdout, f)
	return err
}

func runWith(t *testing.T, change func(*System), args ...string) result {
	t.Helper()
	var stdout, stderr bytes.Buffer
	sys := System{
		Stdin:            strings.NewReader("\n"),
		Stdout:           &stdout,

		StdinIsTerminal:  true,
		StderrIsTerminal: true,
	}
	change(&sys)
	code := Run(args, sys)
	return result{code: code, stdout: stdout.String(), stderr: stderr.String()}
}

// run roda o janus com args e o System de teste padrão.
func run(t *testing.T, args ...string) result {

// saída de erro (FR-CLI-12, NFR-CLI-01). A frase não é conferida.
func (r result) wantFailure(t *testing.T, code int, items ...string) {
	t.Helper()
	if r.code != code {
		t.Errorf("code %d, want %d (stderr %q)", r.code, code, r.stderr)
	}

		t.Errorf("stdout %q, want empty", r.stdout)
	}
	for _, item := range items {
		if !strings.Contains(r.stderr, item) {
			t.Errorf("stderr %q does not cite %q", r.stderr, item)
		}
	}
}

package cli

import (
	"errors"

	"gitlab.com/rg-dev/janus/internal/core"
	"gitlab.com/rg-dev/janus/internal/failure"
)

// Códigos de saída do contrato (docs/specs/bootstrap.md, "Códigos de saída").
const (
	codeOK             = 0
	codeUnexpected     = 1
	codeUsage          = 2
	codeTemplatesDir   = 3
	codeManifest       = 4
	codeNotFound       = 5
	codeNameRequired   = 6
	codeInvalidName    = 7
	codeExists         = 8
	codeMissingProgram = 9
	codeDestinationDir = 10
	codeCanceled       = 130
)

// exitCode traduz err no código de saída (FR-CLI-10). O manifesto vem antes do nome
// inválido porque um nome fixo inválido chega embrulhado num ManifestError.
func exitCode(err error) int {
	switch {
	case err == nil:
		return codeOK
	case errors.Is(err, failure.ErrCanceled):
		return codeCanceled
	case errors.As(err, new(*failure.UsageError)):
		return codeUsage
	case errors.As(err, new(*core.TemplatesDirError)):
		return codeTemplatesDir
	case errors.As(err, new(*core.ManifestError)):
		return codeManifest
	case errors.As(err, new(*core.TemplateNotFoundError)), errors.As(err, new(*failure.NoTemplatesError)):
		return codeNotFound
	case errors.As(err, new(*core.NameRequiredError)):
		return codeNameRequired
	case errors.As(err, new(*core.InvalidNameError)):
		return codeInvalidName
	case errors.As(err, new(*core.DestinationExistsError)):
		return codeExists
	case errors.As(err, new(*failure.MissingProgramError)):
		return codeMissingProgram
	case errors.As(err, new(*core.DestinationDirError)):
		return codeDestinationDir
	}
	return codeUnexpected
}

	"gitlab.com/rg-dev/janus/internal/failure"
	"gitlab.com/rg-dev/janus/internal/interactive"
	"gitlab.com/rg-dev/janus/internal/messages"
)

// Run executa o janus com args, os argumentos sem o nome do programa, e devolve o código
// de saída.
func Run(args []string, sys System) int {
	if len(args) > 0 && !strings.HasPrefix(args[0], "-") {
		return finish(sys, runCommand(args[0], args[1:], sys), false)
	}
	p, err := parseArgs("", args, map[string]bool{"hold": false, "help": false, "version": false})
	switch {
	case err != nil:
		return finish(sys, err, false)
	case p.has("help"):
		return finish(sys, printAlone(sys, "", "--help", args, messages.Help), false)
	case p.has("version"):
		return finish(sys, printAlone(sys, "", "--version", args, version()+"\n"), false)
	case len(p.args) > 0:
		return finish(sys, &failure.UsageError{Problem: failure.UnexpectedArgument, Arg: p.args[0]}, false)
	}
	return finish(sys, runInteractive(sys), p.has("hold"))
}

// runCommand roda um subcomando.
func runCommand(name string, args []string, sys System) error {
	switch name {
	case "list":
		return runList(args, sys)
	case interactive.PreviewCommand:
		return runPreview(args, sys)
	}
	return &failure.UsageError{Problem: failure.UnknownCommand, Arg: name}
}

// printAlone imprime text quando option é o único argumento. Acompanhada de outros, ela é
// uso incorreto (FR-CLI-01).
func printAlone(sys System, command, option string, args []string, text string) error {
	for _, arg := range args {
		if arg != option {
			return &failure.UsageError{Problem: failure.UnexpectedArgument, Arg: arg, Command: command}
		}
	}
	_, err := io.WriteString(sys.Stdout, text)
	return err
}

// version devolve a versão que o Go grava no binário, a partir do git (DE-22).
func version() string {
	if info, ok := debug.ReadBuildInfo(); ok && info.Main.Version != "" {
		return info.Main.Version
	}
	return "(devel)"
}

// finish mostra o erro, quando há, e devolve o código de saída. Cancelar não mostra nada.
// Com hold, espera um Enter depois do erro, para o yazi não esconder a mensagem (FR-TUI-13).
func finish(sys System, err error, hold bool) int {
	code := exitCode(err)
	if code == codeOK || code == codeCanceled {
		return code
	}
	fmt.Fprintln(sys.Stderr, messages.ErrorLine(err))
	if hold {
		fmt.Fprint(sys.Stderr, messages.HoldPrompt)
		_, _ = bufio.NewReader(sys.Stdin).ReadString('\n')
	}
	return code
}

// runList roda `janus list [--json]` (FR-CLI-05, FR-CLI-06). A ordem de verificação é 2,
// 3, 4, 1 (FR-CLI-11).
func runList(args []string, sys System) error {
	p, err := parseArgs("list", args, map[string]bool{"json": false, "help": false})
	if err != nil {
		return err
	}
	if p.has("help") {
		return printAlone(sys, "list", "--help", args, messages.ListHelp)
	}
	if len(p.args) > 0 {
		return &failure.UsageError{Problem: failure.UnexpectedArgument, Arg: p.args[0], Command: "list"}
	}
	cat, err := loadCatalog(sys)
	if err != nil {
		return err
	}
	if p.has("json") {
		return writeJSON(sys.Stdout, cat.Templates)
	}
	var b strings.Builder
	for _, t := range cat.Templates {
		b.WriteString(t.ID)
		b.WriteByte('\n')
	}
	_, err = io.WriteString(sys.Stdout, b.String())
	return err
}

// loadCatalog descobre a pasta de modelos e a lê (códigos 3 e 4).
func loadCatalog(sys System) (core.Catalog, error) {
	dir, err := core.FindTemplatesDir(sys.Getenv)
	if err != nil {
		return core.Catalog{}, err
	}
	return core.LoadCatalog(dir)
}

// jsonTemplate é um item de `janus list --json`. Os nomes das chaves são contrato.
type jsonTemplate struct {
	ID       string   `json:"id"`
	Category string   `json:"category"`
	File     string   `json:"file"`
	Name     jsonName `json:"name"`
}

type jsonName struct {
	Mode       string `json:"mode"`
	Value      string `json:"value,omitempty"`      // só em fixed
	Suggestion string `json:"suggestion,omitempty"` // só em ask
}

// writeJSON escreve a lista do contrato (FR-CLI-06).
func writeJSON(w io.Writer, templates []core.Template) error {
	items := make([]jsonTemplate, 0, len(templates)) // vazia sai como [], e não como null
	for _, t := range templates {
		name := jsonName{Mode: string(t.Mode)}
		if t.Mode == core.NameFixed {
			name.Value = t.FixedName
		} else {
			name.Suggestion = t.File
		}
		items = append(items, jsonTemplate{ID: t.ID, Category: t.Category, File: t.File, Name: name})
	}
	enc := json.NewEncoder(w)
	enc.SetEscapeHTML(false) // "<", ">" e "&" saem como são
	enc.SetIndent("", "  ")
	return enc.Encode(items)
}

		return err
	}
	_, err = fmt.Fprintln(sys.Stdout, path)
	return err
}

// sobrescreve. As verificações seguem a ordem do contrato: 2, 3, 4, 5, 10, 6, 7, 8 e 1
// (FR-CLI-11).
func runNew(args []string, sys System) error {
	p, err := parseArgs("new", args, map[string]bool{"dir": true, "name": true, "help": false})
	if err != nil {
		return err
	}
	if p.has("help") {
		return printAlone(sys, "new", "--help", args, messages.NewHelp)
	}
	switch {
	case len(p.args) == 0:
		return &failure.UsageError{Problem: failure.MissingArgument, Arg: "<id>", Command: "new"}

		return &failure.UsageError{Problem: failure.UnexpectedArgument, Arg: p.args[1], Command: "new"}
	}

	if err != nil {
		return err
	}
	tpl, err := cat.Find(p.args[0])
	if err != nil {
		return err
	}
	dir, hasDir := p.options["dir"]
	if !hasDir {
		if dir, err = sys.Getwd(); err != nil {
			return err
		}
	}
	destDir, err := core.DestinationDir(dir)
	if err != nil {
		return err
	}
	name, hasName := p.options["name"]
	finalName, err := core.FinalName(tpl, name, hasName)
	if err != nil {
		return err
	}
	path, err := core.Create(tpl, destDir, finalName)
	if err != nil {
		return err
	}
	_, err = fmt.Fprintln(sys.Stdout, path)
	return err
}

package cli

import (
	"os"
	"syscall"
	"unsafe"
)

// isTerminal diz se f é um terminal, pedindo ao sistema os atributos de terminal (TCGETS),
// que só um terminal tem. os.ModeCharDevice foi rejeitado porque /dev/null também é um
// dispositivo de caractere. O projeto é só para Linux, e o syscall da biblioteca padrão
// dispensa golang.org/x/term.
func isTerminal(f *os.File) bool {
	var attrs syscall.Termios
	_, _, errno := syscall.Syscall(syscall.SYS_IOCTL, f.Fd(), syscall.TCGETS, uintptr(unsafe.Pointer(&attrs)))
	return errno == 0
}

// System reúne o que o programa recebe do sistema operacional. O main usa OSSystem; os

type System struct {

	StderrIsTerminal bool

#!/usr/bin/env bash
# Conferência manual de RM-01.01 com o fzf real, num tmux (TK-18).
set -u
W=${1:?pasta de trabalho}
JANUS="$W/pasta com espaço/janus"
T=(tmux -L janus-manual -f /dev/null)

# cenario NOME MODELOS ANTES COMANDO TECLAS...: numa pasta de destino vazia, cria o arquivo
# vazio ANTES ("-" para nenhum) e roda COMANDO com XDG_TEMPLATES_DIR=MODELOS, o fish como
# $SHELL e um FZF_DEFAULT_OPTS hostil; depois manda as teclas. "snap" guarda a tela naquele
# momento e "wait" espera meio segundo. O comando vai num script de sh, para não depender
# do shell que o tmux usa. A saída de erro fica no terminal, como o modo interativo exige.
cenario() {
  local name=$1 templates=$2 before=$3 cmd=$4
  shift 4
  rm -rf "$W/dest" && mkdir -p "$W/dest"
  [ "$before" != - ] && : > "$W/dest/$before"
  rm -f "$W/$name.out" "$W/$name.code" "$W/$name.screen"
  cat > "$W/$name.sh" <<EOF
cd '$W/dest'
export SHELL=/usr/bin/fish XDG_TEMPLATES_DIR='$templates'
export FZF_DEFAULT_OPTS='--multi --print-query --expect=ctrl-x'
$cmd > '$W/$name.out'
echo \$? > '$W/$name.code'
sleep 2
EOF
  "${T[@]}" kill-server 2>/dev/null
  "${T[@]}" new-session -d -x 150 -y 30 "sh '$W/$name.sh'"
  sleep 1
  for key in "$@"; do
    case $key in
      snap) "${T[@]}" capture-pane -p >> "$W/$name.screen"; echo ---- >> "$W/$name.screen" ;;
      wait) sleep 0.5 ;;
      *) "${T[@]}" send-keys "$key"; sleep 0.4 ;;
    esac
  done
  sleep 1
  "${T[@]}" kill-server 2>/dev/null
  printf '== %s: código [%s] saída [%s] pasta [%s]\n' "$name" \
    "$(cat "$W/$name.code" 2>/dev/null)" "$(cat "$W/$name.out" 2>/dev/null)" "$(ls -A "$W/dest" | tr '\n' ' ')"
}

Q="'$JANUS'"
cenario fixo        "$W/modelos"  -         "$Q"         react.AGENTS wait Enter
cenario existe      "$W/modelos"  AGENTS.md "$Q"         react.AGENTS wait Enter wait snap C-d
cenario pergunta    "$W/modelos"  -         "$Q"         "Civ. Cont" wait Enter wait snap C-u "Fulano ção" Enter
cenario invalido    "$W/modelos"  -         "$Q"         "Civ. Cont" wait Enter wait C-u a/b Enter wait snap C-u ok Enter
cenario previa      "$W/modelos"  -         "$Q"         fatiar wait snap C-u react.AGENTS wait snap Escape
cenario cancela     "$W/modelos"  -         "$Q"         Escape
cenario ctrl-c-nome "$W/modelos"  -         "$Q"         "Civ. Cont" wait Enter wait C-c
cenario hold        "$W/invalido" -         "$Q --hold"  wait snap Enter
cenario fish        "$W/modelos"  -         "fish '$W/captura.fish' $Q" react.AGENTS wait Enter
```
