```go
// DestinationDirReason diz por que a pasta de destino foi recusada.
type DestinationDirReason int

const (
	DestNotDir          DestinationDirReason = iota + 1 // inexistente ou que não é pasta
	DestInsideTemplates                                 // a pasta de modelos ou uma pasta dentro dela (DE-23)
)

// DestinationDirError é a pasta de destino recusada (código 10): inexistente, que não é
// pasta, ou a pasta de modelos e o que fica dentro dela.
type DestinationDirError struct {
	Dir          string // a pasta como foi informada
	Reason       DestinationDirReason
	TemplatesDir string // a pasta de modelos, só em DestInsideTemplates
	Err          error
}

func (e *DestinationDirError) Error() string {
	if e.Reason == DestInsideTemplates {
		return fmt.Sprintf("destination dir %q is inside the templates dir %q", e.Dir, e.TemplatesDir)
	}
	return fmt.Sprintf("destination dir %q not found", e.Dir)
}

func (e *DestinationDirError) Unwrap() error { return e.Err }

// DestinationExistsError é o destino que já existe (código 8).
type DestinationExistsError struct {
	Path string
}

func (e *DestinationExistsError) Error() string {
	return fmt.Sprintf("destination %s already exists", e.Path)
}

// DestinationDir confere a pasta de destino e devolve o caminho absoluto dela, sem resolver
// os links, para o caminho impresso ser o que o usuário vê. Um link para pasta é aceito
// (FR-CORE-18). O caminho vazio é recusado antes do filepath.Abs, que o trataria como a
// pasta atual. A pasta de modelos e o que fica dentro dela são recusados (FR-CORE-24): a
// verificação mora aqui porque `new` e o modo interativo já passam por esta função, cada
// um no ponto que a ordem do contrato pede (DE-26).
func DestinationDir(dir, templatesDir string) (string, error) {
	notDir := func(err error) error {
		return &DestinationDirError{Dir: dir, Reason: DestNotDir, Err: err}
	}
	if dir == "" {
		return "", notDir(nil)
	}
	abs, err := filepath.Abs(dir)
	if err != nil {
		return "", notDir(err)
	}
	info, err := os.Stat(abs)
	if err != nil {
		return "", notDir(err)
	}
	if !info.IsDir() {
		return "", notDir(nil)
	}
	resolved, err := filepath.EvalSymlinks(abs)
	if err != nil {
		return "", notDir(err)
	}
	inside, err := within(resolved, templatesDir)
	if err != nil {
		return "", err
	}
	if inside {
		return "", &DestinationDirError{Dir: dir, Reason: DestInsideTemplates, TemplatesDir: templatesDir}
	}
	return abs, nil
}

// within diz se realDir, um caminho já sem links, é root ou fica dentro dela. Os links de
// root são resolvidos porque ~/Modelos costuma ser um link (DE-23). O filepath.Rel evita o
// falso positivo da comparação por prefixo de texto, que acharia Modelos-antigo dentro de
// Modelos. Uma montagem com mount --bind escapa desta comparação; os.SameFile subindo até
// a raiz pegaria esse caso, mas custa um Stat por nível e foi rejeitado (DE-26). Uma falha
// ao resolver root, que já foi validada na descoberta, sai como erro inesperado.
func within(realDir, root string) (bool, error) {
	realRoot, err := filepath.EvalSymlinks(root)
	if err != nil {
		return false, err
	}
	rel, err := filepath.Rel(realRoot, realDir)
	if err != nil {
		return false, err
	}
	up := ".." + string(filepath.Separator)
	return rel == "." || (rel != ".." && !strings.HasPrefix(rel, up)), nil
}
```

````markdown
# Janus no yazi

Atalho do [yazi](https://yazi-rs.github.io) que abre o modo interativo do Janus na pasta em que o yazi está: escolha um modelo, confirme o nome, e o arquivo aparece na listagem ao voltar.

## Requisitos

- `janus` no `PATH` que o yazi enxerga.
- `fzf`, que o modo interativo usa para escolher o modelo e pedir o nome.
- `bat`, opcional, para realçar a prévia.

## Instalação

1. Na raiz do repositório do Janus, instale o binário:

   ```
   go install ./cmd/janus
   ```

   Ele vai para `~/go/bin`, que precisa estar no `PATH`.

2. Leve o bloco de [`keymap.toml`](keymap.toml) para `~/.config/yazi/keymap.toml`. Se o arquivo ainda não existe, copie-o inteiro:

   ```
   mkdir -p ~/.config/yazi
   cp integrations/yazi/keymap.toml ~/.config/yazi/keymap.toml
   ```

   Se ele já existe, acrescente o bloco `[[mgr.prepend_keymap]]` ao fim, sem apagar os outros.

3. Feche e abra o yazi de novo: ele só lê o `keymap.toml` na partida.

## Uso

Na pasta onde o arquivo deve nascer, tecle `c` e depois `m`.

- Um modelo com nome fixo é criado direto; os outros pedem o nome, sugerindo o do modelo.
- Se o arquivo já existe, o campo de nome abre com um aviso, para você escolher outro nome.
- `Esc` cancela e volta ao yazi sem criar nada.
- Um erro, como um manifesto inválido, fica na tela até o `Enter`.
- Dentro da pasta de modelos o atalho recusa criar o arquivo: o Janus nunca escreve nela.

As outras teclas do `c`, que copiam caminhos, continuam funcionando: o menu do `c` mostra todas.

## Problemas comuns

- **O menu do `c` não mostra "Novo arquivo a partir de modelo":** o yazi não leu o bloco. Confira o caminho do arquivo, ou a pasta de `YAZI_CONFIG_HOME`, se ela estiver definida, e reinicie o yazi.
- **A tela pisca e volta ao yazi sem fazer nada:** o `janus` não está no `PATH` que o yazi enxerga, e a mensagem do shell some junto com a tela. Acontece quando o yazi é aberto por um lançador gráfico que não leva o `~/go/bin` no `PATH`. Abra o yazi pelo terminal, ou ponha o `~/go/bin` no `PATH` da sessão gráfica.
````

```bash
#!/usr/bin/env bash
# Conferência manual de RM-01.02 com o yazi real, num tmux (TK-22).
set -u
W=${1:?pasta de trabalho}
REPO=${2:?raiz do repositório}
T=(tmux -L janus-yazi -f /dev/null)

# sessao NOME BIN MODELOS PASTA ACOES...: abre o yazi em PASTA, com o keymap entregue,
# PATH=BIN:/usr/bin:/bin e XDG_TEMPLATES_DIR=MODELOS, e executa as ações: "snap" guarda a
# tela, "ls" guarda o conteúdo de PASTA, "wait" espera um segundo, "txt:TEXTO" digita
# TEXTO sem interpretar nomes de tecla, e o resto vai como tecla do tmux. No fim, sai do
# yazi com q. O comando vai num script de sh, para não depender do shell que o tmux usa.
sessao() {
  local name=$1 bin=$2 templates=$3 dir=$4
  shift 4
  local log="$W/$name.log"
  : > "$log"
  cat > "$W/$name.sh" <<EOF
cd '$dir'
export PATH='$bin':/usr/bin:/bin XDG_TEMPLATES_DIR='$templates'
export YAZI_CONFIG_HOME='$REPO/integrations/yazi' XDG_STATE_HOME='$W/state' XDG_CACHE_HOME='$W/cache'
exec yazi
EOF
  "${T[@]}" kill-server 2>/dev/null
  "${T[@]}" new-session -d -x 150 -y 35 "sh '$W/$name.sh'"
  sleep 2
  for a in "$@"; do
    case $a in
      snap) { echo "---- tela"; "${T[@]}" capture-pane -p; } >> "$log" ;;
      ls) { echo "---- pasta: $(ls -A "$dir" | tr '\n' '|')"; } >> "$log" ;;
      wait) sleep 1 ;;
      txt:*) "${T[@]}" send-keys -l "${a#txt:}"; sleep 0.5 ;;
      *) "${T[@]}" send-keys "$a"; sleep 0.5 ;;
    esac
  done
  "${T[@]}" send-keys q
  sleep 1
  "${T[@]}" kill-server 2>/dev/null
  echo "======== $name"
  cat "$log"
}

B="$W/bin"
sessao passos-1-a-4 "$B" "$W/Modelos" "$W/dest" \
  c m wait snap txt:react.AGENTS wait Enter wait snap ls \
  c m wait txt:react.AGENTS wait Enter wait snap Escape wait snap ls \
  c m wait "txt:Civ. Cont" wait Enter wait snap C-u txt:Fulano Enter wait snap ls
sessao passo-5 "$B" "$W/invalido" "$W/dest-invalido" c m wait wait snap Enter wait snap ls
touch "$W/marca"
sleep 1
sessao passo-6 "$B" "$W/Modelos" "$W/Modelos/info.ai" c m wait wait snap Enter wait snap
echo "======== escrito na pasta de modelos depois da marca:"
find "$W/modelos-copia" -newer "$W/marca" -print
sessao passo-7 "$B" "$W/Modelos" "$W/dest" c wait snap Escape wait snap
rm -rf "$W/dest-sem-janus" && mkdir "$W/dest-sem-janus"
sessao sem-janus "$W/vazio" "$W/Modelos" "$W/dest-sem-janus" c m wait snap wait snap ls
```
